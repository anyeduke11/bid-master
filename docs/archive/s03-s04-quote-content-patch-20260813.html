# ====== BidBoard 集成(末尾追加) ======
# 接受 --bid-code 参数(从命令行传入客户代号,默认空)
import argparse as _argparse
_bid_parser = _argparse.ArgumentParser(add_help=False)
_bid_parser.add_argument("--bid-code", default="", help="客户代号(例 dgyhst2026)")
_bid_parser.add_argument("--quote-version", default="v1", help="报价版本号")
_bid_args, _ = _bid_parser.parse_known_args()

if _bid_args.bid_code:
    import os, sys
    sys.path.insert(0, os.path.join(os.path.dirname(__file__)))
    from bidboard_client import add_ai_ready, set_stage, set_block, set_human_todo
    CODE = _bid_args.bid_code
    VER = _bid_args.quote_version

    # 集成点 A:BOM 索引加载完(在 main() 调 bom_extract 之后)
    # (调用方在调用 main() 前确保 BOM 已加载;此处仅在 main 末尾追加)

    # 集成点 B:报价单 v1 出(在 quote_record_manager.save 之后)
    # 调用方负责传 --quote-version=v1 / v2 / final
    if VER in ("v1", "v2", "v3", "final"):
        add_ai_ready(CODE, f"报价单 {VER} (折扣 {getattr(_bid_args, 'discount', '1.0')})", "quote-generation-v3")
        set_stage(CODE, "报价")

    # 集成点 C:等签字(报价单完成,Duke 需签字)
    set_block(CODE, "报价签字")
    set_human_todo(CODE, ["报价签字", "确认折扣率"])

    # 签字后手动解除(主流程外):
    #   python3 -c "from bidboard_client import set_block, set_stage; set_block('dgyhst2026', ''); set_stage('dgyhst2026', '修订')"

说明:真实集成点 A/B 应该在 generate_quote_v3.py 的 main() 函数内部(如 bom_search 后 / quote_record_manager.save 后),而不是 main 末尾。上面示例是简化版,真实 patch 需要在 main 函数里精准插桩。

## 八、BidBoard 集成(必须)

报价单生成的 3 个关键节点自动调 BidBoard API 写看板。

### 集成点 A · BOM 索引加载完

```python
from bidboard_client import add_ai_ready
```

### 集成点 B · 报价单 v1 出
在 `quote_record_manager.save()` 后,自动调:
```python
from bidboard_client import add_ai_ready, set_stage
add_ai_ready(CODE, f"报价单 {VERSION} (折扣 {DISCOUNT})", "quote-generation-v3")
set_stage(CODE, "报价")
```

### 集成点 C · 等签字
报价单完成 → 自动阻塞,等 Duke 签字:

```python
from bidboard_client import set_block, set_human_todo
```

# ====== BidBoard 集成(末尾追加) ======
import argparse as _argparse
_bid_parser = _argparse.ArgumentParser(add_help=False)
_bid_parser.add_argument("--bid-code", default="", help="客户代号")
_bid_parser.add_argument("--word-count", type=int, default=0, help="草稿字数")
_bid_args, _ = _bid_parser.parse_known_args()

if _bid_args.bid_code:
    import os, sys
    sys.path.insert(0, os.path.join(os.path.dirname(__file__)))
    from bidboard_client import add_ai_ready, set_human_todo
    CODE = _bid_args.bid_code

    # 集成点 A:草稿完成(在 main 写完草稿后)
    if _bid_args.word_count > 0:
        add_ai_ready(CODE, f"长文草稿 {_bid_args.word_count} 字", "content-longform")

    # 集成点 B:品牌自检通过(在 brand-voice-compliance.py 跑完后调)
    # 调用方式:python3 longform_draft.py --bid-code=xxx --word-count=N --brand-checked
    if getattr(_bid_args, 'brand_checked', False):
        add_ai_ready(CODE, "品牌自检通过", "content-longform")
        set_human_todo(CODE, ["终审", "发布或定稿"])

说明:实际集成时,把上面 addAiReady 调用精准插到 longform_draft.py 写完草稿的那一行后(不是 main 末尾)。brand_checked 标志位由调用方在 brand-voice-compliance 通过后传入。
