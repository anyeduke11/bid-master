#!/usr/bin/env python3
"""
BidBoard 客户端 · 所有 Skill 复用
- 6 个公开 API(create / add_ai_ready / set_stage / set_block / set_human_todo / patch)
- 失败兜底:写入 ~/.bidboard_queue.jsonl,下次启动重放
- 单文件零依赖(只 stdlib)
"""
import os, json, urllib.request, urllib.error
from datetime import datetime
from pathlib import Path

BOARD_URL = os.environ.get("BIDBOARD_URL", "http://127.0.0.1:8765")
TOKEN     = os.environ.get("BIDBOARD_TOKEN", "")
QUEUE     = Path(os.environ.get("BIDBOARD_QUEUE", str(Path.home() / ".bidboard_queue.jsonl")))
TIMEOUT   = 3  # 秒


def _post(path: str, payload: dict) -> dict:
    """POST 请求 + 失败兜底入队列。返回 {ok, data} 或 {ok: False, queued: True}。"""
    url = BOARD_URL + path
    body = json.dumps(payload, ensure_ascii=False).encode("utf-8")
    headers = {
        "Content-Type": "application/json",
        "Authorization": f"Bearer {TOKEN}",
    }
    try:
        req = urllib.request.Request(url, data=body, headers=headers, method="POST")
        with urllib.request.urlopen(req, timeout=TIMEOUT) as r:
            return {"ok": True, "data": json.loads(r.read().decode("utf-8"))}
    except (urllib.error.URLError, TimeoutError, json.JSONDecodeError, OSError) as e:
        # 失败兜底:写入本地队列,下次启动时 workbench 重放
        _enqueue(path, payload, str(e))
        return {"ok": False, "queued": True, "err": str(e)}


def _enqueue(path: str, payload: dict, err: str):
    try:
        QUEUE.parent.mkdir(parents=True, exist_ok=True)
        with open(QUEUE, "a", encoding="utf-8") as f:
            f.write(json.dumps({
                "ts": datetime.now().isoformat(timespec="seconds"),
                "path": path, "payload": payload, "err": err
            }, ensure_ascii=False) + "\n")
    except OSError:
        pass  # 队列也写不进去,放弃


# ============ 公开 API ============

def create(code: str, client: str, **kw) -> dict:
    """新商机入 bids。code 必须唯一(11 家在跑客户的代号风格)。"""
    return _post("/api/bids", {"code": code, "client": client, **kw})


def add_ai_ready(code: str, item: str, source: str = "") -> dict:
    """Skill 跑完追加 ai_ready 项(自动去重)。"""
    return _post(f"/api/bids/{code}/ai_ready", {"value": item, "source": source})


def set_stage(code: str, stage: str) -> dict:
    """推进客户阶段(枚举:跟踪/审查/报价/修订/封标/述标)。"""
    return _post(f"/api/bids/{code}/stage", {"value": stage})


def set_block(code: str, reason: str) -> dict:
    """写入阻塞原因(空字符串表示解除)。"""
    return _post(f"/api/bids/{code}/block", {"value": reason})


def set_human_todo(code: str, todos: list) -> dict:
    """写入"人要做的事"列表。"""
    return _post(f"/api/bids/{code}/human_todo", {"value": todos})


def set_due_at(code: str, due_at: str) -> dict:
    """更新截止日(格式:YYYY-MM-DD HH:MM)。"""
    return _post(f"/api/bids/{code}/due_at", {"value": due_at})


def update_flow_node(code: str, node_id: str, status: str, block_reason: str = "") -> dict:
    """更新流程节点状态。status: done/progress/blocked/planned/pending。"""
    return _post(f"/api/bids/{code}/flow/{node_id}",
        {"status": status, "block_reason": block_reason})


# ============ 工具:重放队列(workbench 启动时跑) ============

def replay_queue(max_n: int = 100) -> dict:
    """重放 ~/.bidboard_queue.jsonl。成功一条删一条,3 次失败转死信。"""
    if not QUEUE.exists():
        return {"replayed": 0, "dead": 0}
    dead = QUEUE.parent / ".bidboard_dead.jsonl"
    survived = []
    ok_n = dead_n = 0
    with open(QUEUE, "r", encoding="utf-8") as f:
        lines = f.readlines()[-max_n:]
    for ln in lines:
        try:
            rec = json.loads(ln)
            r = _post(rec["path"], rec["payload"])
            if r.get("ok"):
                ok_n += 1
            else:
                rec["retry"] = rec.get("retry", 0) + 1
                if rec["retry"] >= 3:
                    with open(dead, "a", encoding="utf-8") as f:
                        f.write(ln)
                    dead_n += 1
                else:
                    survived.append(json.dumps(rec, ensure_ascii=False))
        except Exception:
            survived.append(ln.strip())
    with open(QUEUE, "w", encoding="utf-8") as f:
        f.write("\n".join(survived) + ("\n" if survived else ""))
    return {"replayed": ok_n, "dead": dead_n, "survived": len(survived)}


# ============ CLI 测试入口 ============
if __name__ == "__main__":
    import sys
    if len(sys.argv) > 1 and sys.argv[1] == "replay":
        print(replay_queue())
    else:
        print("用法: python3 bidboard_client.py replay")

#!/usr/bin/env python3
"""
post_collect_sync.py · bid-news-collection 收尾同步
用法: python3 post_collect_sync.py --bids /path/to/bids.json [--dry-run]

读 bids.json(gen_md.py 的输入),逐条识别:
  1) bidder 在 known_clients → 老项目 → setStage(审查) + setDueAt
  2) bidder 不在 known_clients + type_tag='🆕新增' → 新商机 → create
  3) 其他(已结果公示 / 已截止) → 仅 add_ai_ready 标记,不入 bids
"""
import argparse, json, sys, re
from datetime import datetime, date
from pathlib import Path

SKILL_DIR = Path(__file__).resolve().parent.parent


def load_config():
    p = SKILL_DIR / "config.json"
    if p.exists():
        return json.loads(p.read_text(encoding="utf-8"))
    return {}


def _gen_code(client_name: str, today: date) -> str:
    """生成 6 位代号:客户拼音首字母 + 年月日(取 6 位)。"""
    # 简化:取客户名前 4 个字符(pinyin 或汉字)+ 年月后 2 位
    py_map = {
        "国信证券": "gxzq", "国泰君安": "gtja", "某证券机构A": "zxzq",
        "华泰证券": "htzq", "招商证券": "zszq", "平安银行": "payh",
        "东莞银行": "dgyh", "农发行": "nfh", "银联国际": "unionpay-intl",
    }
    # 优先查表,否则用 pypinyin / 简化处理
    base = py_map.get(client_name)
    if not base:
        try:
            from pypinyin import lazy_pinyin
            base = "".join(lazy_pinyin(client_name))[:6]
        except ImportError:
            base = re.sub(r"\s+", "", client_name)[:4]  # 无 pypinyin 时退化
    # 6 位代号 = base(变长) + 年月日(取后 4 位)
    suffix = today.strftime("%y%m%d")
    code = f"{base}{suffix}"
    return code[:12]  # 限制长度


def _parse_due(deadline_str: str) -> str:
    """deadline 形如 '2026-05-25(公示)' 或 '2026-06-30 17:00',提取日期。"""
    if not deadline_str:
        return ""
    m = re.search(r"(\d{4}-\d{2}-\d{2})", str(deadline_str))
    if not m:
        return ""
    return f"{m.group(1)} 17:00"


def _calc_priority(due_at: str, today: date) -> str:
    """按 §6.5 优先级算法:T≤3→P0, T≤7→P1, 其他→P2。"""
    if not due_at:
        return "P2"
    try:
        due = datetime.strptime(due_at.split()[0], "%Y-%m-%d").date()
    except (ValueError, IndexError):
        return "P2"
    days = (due - today).days
    if days <= 3:
        return "P0"
    if days <= 7:
        return "P1"
    return "P2"


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--bids", required=True, help="gen_md.py 的输入 bids.json 路径")
    ap.add_argument("--dry-run", action="store_true", help="只打印,不发请求")
    args = ap.parse_args()

    bids_path = Path(args.bids)
    if not bids_path.exists():
        print(f"❌ 找不到 bids 文件: {bids_path}")
        sys.exit(1)

    bids = json.loads(bids_path.read_text(encoding="utf-8")).get("bids", [])
    cfg = load_config()
    known = set(cfg.get("known_clients", []))
    today = date.today()

    # 延迟 import(让 --help / dry-run 不依赖 workbench 起来)
    if not args.dry_run:
        from bidboard_client import create, add_ai_ready, set_stage, set_due_at, set_human_todo
    else:
        create = add_ai_ready = set_stage = set_due_at = set_human_todo = lambda *a, **k: {"ok": True, "dry": True}

    n_new = n_update = n_mark = n_skip = n_fail = 0

    for b in bids:
        bidder = b.get("bidder", "").strip()
        tag = b.get("type_tag", "")
        project = b.get("project", "")[:40]
        due = _parse_due(b.get("deadline"))

        if not bidder:
            n_skip += 1
            continue

        # === A. 老项目(在 known_clients) ===
        if bidder in known:
            code = cfg["client_code_map"].get(bidder, "")
            if not code:
                print(f"⚠ 已知客户但无 code 映射: {bidder},跳过")
                n_skip += 1
                continue
            r1 = set_stage(code, "审查")
            r2 = add_ai_ready(code, f"标讯命中: {project}", "bid-news-collection")
            if due:
                set_due_at(code, due)
            if r1.get("ok") and r2.get("ok"):
                n_update += 1
                print(f"✓ 老项目更新: {code} ({bidder}) — {project}")
            else:
                n_fail += 1
                print(f"✗ 失败: {code} — {r1.get('err') or r2.get('err')}")
            continue

        # === B. 新商机(🆕新增 类型才入 bids) ===
        if "🆕" in tag or "新增" in tag:
            code = _gen_code(bidder, today)
            priority = _calc_priority(due, today)
            r = create(code, bidder,
                due_at = due,
                priority = priority,
                stage = "审查",
                ai_ready = ["标讯命中"],
                human_todo = ["下载招标文件", "判断是否参与"],
                source = "bid-news-collection"
            )
            if r.get("ok"):
                n_new += 1
                print(f"✓ 新商机入 bids: {code} ({bidder}) priority={priority} due={due}")
            else:
                n_fail += 1
                print(f"✗ 入 bids 失败: {code} — {r.get('err')}")
            continue

        # === C. 跳过(已结果 / 已截止 / 政策) ===
        n_mark += 1
        print(f"· 跳过(非新商机): {bidder} — {project}")

    print(f"\n=== 同步摘要 ===")
    print(f"  新商机入 bids: {n_new}")
    print(f"  老项目更新:    {n_update}")
    print(f"  跳过(非新商机): {n_mark}")
    print(f"  失败:          {n_fail}")
    print(f"  跳过(无 bidder): {n_skip}")
    if args.dry_run:
        print("  (DRY RUN — 未实际发送请求)")


if __name__ == "__main__":
    main()

## Step 8 · BidBoard 同步(必须 · 新)

采集完成后,自动(或手动)把结果推送到投标看板。

### 8.1 触发方式

- **自动**:`run.py` 在末尾检测到 `~/WorkBuddy/Claw/bids_YYYYMMDD.json` → 调 `post_collect_sync.py`

### 8.2 同步逻辑

```
对每条 bid:
  1) bidder 在 known_clients → 老项目
     - setStage(code, "审查")
     - addAiReady(code, "标讯命中: ")
     - setDueAt(code, parsed_deadline)
  2) bidder 不在 known_clients + type_tag 含 "🆕新增" → 新商机
     - 生成 code: 客户拼音 + YYMMDD
     - calcPriority(due): T≤3→P0, T≤7→P1, 其他→P2

  3) 其他(✅结果 / ⚠️已截止 / 🔴政策)
     - 仅 print 跳过,不调 API
```

7风险与回滚
🔴 pypinyin 依赖

_gen_code 默认用 pypinyin 转换中文客户名,未装时退化但 code 质量差。
