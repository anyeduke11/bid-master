#!/usr/bin/env python3
"""
review_sync.py · bid-file-review 收尾同步
用法:
  # 5 类场景,任意时机调一次
  python3 review_sync.py --stage audit        --code dgyhst2026
  python3 review_sync.py --stage ocr          --code dgyhst2026 --count 12
  python3 review_sync.py --stage review       --code dgyhst2026
  python3 review_sync.py --stage report       --code dgyhst2026 --output /path/to/报告.md
  python3 review_sync.py --stage critical     --code dgyhst2026 --reason "缺 ★条款响应"
  python3 review_sync.py --stage done         --code dgyhst2026  # 全部完成,推 stage=修订

每个 --stage 对应一个 BidBoard API 调用 + 一条事件日志。
"""
import argparse, json, sys
from pathlib import Path

SKILL_DIR = Path(__file__).resolve().parent.parent


def main():
    ap = argparse.ArgumentParser(
        description="bid-file-review → BidBoard 同步",
        formatter_class=argparse.RawDescriptionHelpFormatter
    )
    ap.add_argument("--code", required=True, help="客户代号,如 dgyhst2026")
    ap.add_argument("--stage", required=True,
        choices=["audit", "ocr", "review", "report", "critical", "done"],
        help="评审阶段")
    ap.add_argument("--count", type=int, help="OCR 提取张数(仅 --stage ocr 用)")
    ap.add_argument("--reason", help="废标原因(仅 --stage critical 用)")
    ap.add_argument("--output", help="报告文件路径(仅 --stage report 用)")
    ap.add_argument("--dry-run", action="store_true", help="只打印,不发请求")
    args = ap.parse_args()

    # 延迟 import
    if not args.dry_run:
        from bidboard_client import (
            add_ai_ready, set_stage, set_block, set_human_todo
        )
    else:
        # dry-run 模式不调 API
        class _Stub:
            def __getattr__(self, name):
                return lambda *a, **k: {"ok": True, "dry": True, "call": name}
        add_ai_ready = set_stage = set_block = set_human_todo = _Stub()

    results = []
    code = args.code

    if args.stage == "audit":
        # 阶段零:招标审计通过
        r = add_ai_ready(code, "招标审计通过(15 项)", "bid-file-review")
        results.append(("audit", r))

    elif args.stage == "ocr":
        # 步骤二:OCR 强制提取完成
        count = args.count or 0
        r = add_ai_ready(code, f"OCR 提取 {count} 张图片", "bid-file-review")
        results.append(("ocr", r))

    elif args.stage == "review":
        # 步骤三~六:P1~P7 评审完
        r = add_ai_ready(code, "100+ 项检查完成(P1~P7)", "bid-file-review")
        results.append(("review", r))

    elif args.stage == "report":
        # 步骤七:最终报告输出
        output = args.output or ""
        item = f"评审报告已生成"
        if output:
            item += f": {Path(output).name}"
        r = add_ai_ready(code, item, "bid-file-review")
        results.append(("report", r))
        # 报告输出后,把 stage 推"修订",给 Duke 看
        r2 = set_stage(code, "修订")
        results.append(("stage→修订", r2))
        # 提示人工做差异点比对
        r3 = set_human_todo(code, ["审阅评审报告", "差异点比对"])
        results.append(("human_todo", r3))

    elif args.stage == "critical":
        # 检测到废标条款 → 阻塞
        reason = args.reason or "未指定"
        r = set_block(code, f"废标风险: {reason}")
        results.append(("block", r))
        r2 = add_ai_ready(code, f"⚠ 检测到废标条款: {reason}", "bid-file-review")
        results.append(("critical", r2))

    elif args.stage == "done":
        # 全部完成,清理阻塞,推 stage='封标'
        r = set_block(code, "")
        results.append(("clear_block", r))
        r2 = add_ai_ready(code, "评审完成,可进入下一阶段", "bid-file-review")
        results.append(("done", r2))

    # 打印结果
    if args.dry_run:
        print("🔍 DRY RUN — 模拟执行:")
    ok_n = fail_n = 0
    for name, r in results:
        if r.get("ok"):
            ok_n += 1
            if r.get("dry"):
                print(f"  · [DRY] {name}")
            elif r.get("queued"):
                print(f"  ⏳ {name}: 已入队列")
            else:
                print(f"  ✓ {name}: 成功")
        else:
            fail_n += 1
            print(f"  ✗ {name}: {r.get('err')}")

    print(f"\n=== 同步完成: {ok_n} 成功, {fail_n} 失败 ===")
    sys.exit(0 if fail_n == 0 else 1)


if __name__ == "__main__":
    main()

6风险与回滚
🔴 AI 不调 review_sync

SKILL.md 加了指引但 AI 不一定照做。
缓解:每次跑完整标书后,人工抽查事件流(底部)是否出现 bid-file-review 来源;不出现就强制再调。

🟡 --code 传错

AI 拼错客户代号,写入错误的客户卡。
缓解:review_sync 接受任何 code,看板有重复 code 时手动合并。

🔴 critical 误触发

AI 把"扣 5 分"当成"废标",刷出红色告警。
缓解:review_sync 接受 Duke 手动 setBlock('') 解除,误报可秒消。

🔵 OCR 集成点跳过

某些场景 AI 觉得 OCR 没必要做(如投标文件纯文本),跳过 review_sync --stage ocr。
缓解:是 OK 的,只少一项 ai_ready 不影响整体。
