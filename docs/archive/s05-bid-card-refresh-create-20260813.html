# 投标看板每日刷新 · bid-card-refresh

> **定位**:这是"看板的心脏"。每个工作日 09:45 自动跑一次,确保看板数据是最新的、priority 是准的、P0 客户会被及时告警。
> **不是**后台 daemon(那是 p0-watcher 的事),而是"被 cron 调起一次就退出"的脚本。

## 一、职责

每日 09:45 跑一次,执行三件事:
1. **重算 priority** — 按 §6.5 算法(见后),priority 变了才 PATCH(避免无意义写入)
2. **写 last_sync** — 所有 bids 写当前时间,证明看板"活的"
3. **触发 p0-watcher** — 把 P0 客户列表作为参数,spawn p0-watcher Skill

## 二、触发场景

- **自动**(推荐):macOS launchd / Linux crontab 每日 09:45
- **手动**:`python3 scripts/main.py --once`(单次跑)
- **测试**:`python3 scripts/main.py --once --dry-run`(不发请求)
- **强制**:`python3 scripts/main.py --once --force`(即使今天已跑过也跑)

## 三、执行步骤

### 步骤 1:拉取所有 bids

GET `http://127.0.0.1:8765/api/bids` → 列表

### 步骤 2:逐个重算 priority

```
对每条 bid:
    old_priority = bid.priority
    new_priority = calc_priority(bid, today)
    if new_priority != old_priority:
        PATCH /api/bids//priority  {value: new_priority}
        log_event("priority 变化", code, old → new)
```

### 步骤 3:写 last_sync

```
对每条 bid:
    PATCH /api/bids//last_sync  {value: now()}
```

### 步骤 4:触发 P0 看门狗

```
p0_list = [b.code for b in bids if b.priority == "P0"]
if p0_list:
    spawn("p0-watcher", clients=p0_list)  # 通过 workbuddy run / wb_cli
```

### 步骤 5:幂等性

- 同一天多次跑,结果应一致
- 用 `~/.bidboard_refresh_YYYYMMDD` 标记今日已跑
- 强制重跑:`--force`

## 四、§6.5 优先级算法(复述)

```python
def calc_priority(bid, today):
    if not bid.due_at or bid.due_at == "持续":
        return "P2"
    if all nodes done:
        return "P2"
    days = (due - today).days
    if days <= 3: priority = "P0"
    elif days <= 7: priority = "P1"
    else: priority = "P2"
    has_block = any(n.status == "blocked" for n in bid.flow.nodes)
    if has_block:
        if priority == "P1": priority = "P0"
        elif priority == "P2": priority = "P1"
    return priority
```

#!/usr/bin/env python3
"""
bid-card-refresh · 每日 09:45 跑一次的看板刷新
用法:
  python3 main.py --once             # 单次跑(默认)
  python3 main.py --once --dry-run   # 干跑,不发请求
  python3 main.py --once --force     # 强制跑(忽略今日已跑标记)
  python3 main.py --once --no-spawn  # 不触发 p0-watcher
"""
import argparse, json, os, sys, subprocess
from datetime import datetime, date, timedelta
from pathlib import Path

SKILL_DIR = Path(__file__).resolve().parent.parent
LOG_FILE = Path(os.environ.get(
    "BIDBOARD_REFRESH_LOG",
    str(Path.home() / ".bidboard_refresh.log")))


def load_config():
    p = SKILL_DIR / "config.json"
    if p.exists():
        return json.loads(p.read_text(encoding="utf-8"))
    # 默认配置
    return {
        "board_url": "http://127.0.0.1:8765",
        "token_env": "BIDBOARD_TOKEN",
        "today_override": None,
        "spawn_p0_watcher": True,
        "p0_watcher_cmd": "workbuddy run p0-watcher --clients={clients}",
    }


def log(msg: str):
    """写到 LOG_FILE,带时间戳。"""
    ts = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
    line = f"[{ts}] {msg}"
    print(line)
    try:
        with open(LOG_FILE, "a", encoding="utf-8") as f:
            f.write(line + "\n")
    except OSError:
        pass


def calc_priority(bid: dict, today: date) -> str:
    """§6.5 优先级算法:持续→P2;全完成→P2;T≤3→P0;T≤7→P1;其他→P2;+ 阻塞升级。"""
    due_str = bid.get("due_at", "")
    if not due_str or due_str == "持续":
        return "P2"

    # 全完成 → P2
    flow = bid.get("flow") or {}
    nodes = flow.get("nodes", [])
    if nodes and all(n.get("status") == "done" for n in nodes):
        return "P2"

    # 解析 due_at(形如 "2026-08-13 17:00")
    try:
        due = datetime.strptime(due_str.split()[0], "%Y-%m-%d").date()
    except (ValueError, IndexError):
        return "P2"

    days = (due - today).days
    if days <= 3:
        priority = "P0"
    elif days <= 7:
        priority = "P1"
    else:
        priority = "P2"

    # 阻塞升级
    has_block = any(n.get("status") == "blocked" for n in nodes)
    if has_block:
        if priority == "P1":
            priority = "P0"
        elif priority == "P2":
            priority = "P1"

    return priority


def fetch_bids(cfg: dict) -> list:
    """GET /api/bids → 列表。失败返回空列表(主流程不崩)。"""
    import urllib.request, urllib.error
    url = cfg["board_url"] + "/api/bids"
    token = os.environ.get(cfg.get("token_env", "BIDBOARD_TOKEN"), "")
    headers = {"Authorization": f"Bearer {token}"} if token else {}
    try:
        req = urllib.request.Request(url, headers=headers)
        with urllib.request.urlopen(req, timeout=5) as r:
            data = json.loads(r.read().decode("utf-8"))
            if isinstance(data, list):
                return data
            return data.get("bids", data.get(, []))
    except (urllib.error.URLError, TimeoutError, json.JSONDecodeError, OSError) as e:
        log(f"⚠ fetch_bids 失败(不影响主流程): {e}")
        return []


def patch_field(cfg: dict, code: str, field: str, value) -> bool:
    """PATCH /api/bids/<code>/<field>。失败入队列,返回 True/False。"""
    import urllib.request, urllib.error
    url = f"{cfg['board_url']}/api/bids/{code}/{field}"
    token = os.environ.get(cfg.get("token_env", "BIDBOARD_TOKEN"), "")
    body = json.dumps({"value": value}).encode("utf-8")
    headers = {"Content-Type": "application/json"}
    if token:
        headers["Authorization"] = f"Bearer {token}"
    try:
        req = urllib.request.Request(url, data=body, headers=headers, method="PATCH")
        urllib.request.urlopen(req, timeout=3).read()
        return True
    except (urllib.error.URLError, TimeoutError, OSError) as e:
        # 入本地重试队列
        qfile = Path.home() / ".bidboard_queue.jsonl"
        try:
            with open(qfile, "a", encoding="utf-8") as f:
                f.write(json.dumps({
                    "ts": datetime.now().isoformat(timespec="seconds"),
                    "path": f"/api/bids/{code}/{field}",
                    "payload": {"value": value},
                    "err": str(e)
                }, ensure_ascii=False) + "\n")
        except OSError:
            pass
        return False


def spawn_p0_watcher(cfg: dict, p0_list: list, dry_run: bool):
    """触发 p0-watcher Skill(若启用)。"""
    if not cfg.get("spawn_p0_watcher", True):
        log(f"  · spawn_p0_watcher 关闭,跳过")
        return
    if not p0_list:
        log(f"  · 无 P0 客户,跳过 spawn")
        return
    clients = ",".join(p0_list)
    cmd = cfg.get("p0_watcher_cmd", "workbuddy run p0-watcher --clients={clients}").format(clients=clients)
    log(f"  · 触发 p0-watcher: {cmd}")
    if dry_run:
        return
    try:
        subprocess.Popen(cmd, shell=True, stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)
    except Exception as e:
        log(f"  ⚠ spawn 失败(不影响主流程): {e}")


def check_idempotency(today: date, force: bool) -> bool:
    """返回 True 表示"今天已跑过,应跳过"。force=True 强制跑。"""
    if force:
        return False
    marker = Path.home() / f".bidboard_refresh_{today.strftime('%Y%m%d')}"
    if marker.exists():
        return True
    return False


def mark_done(today: date):
    marker = Path.home() / f".bidboard_refresh_{today.strftime('%Y%m%d')}"
    marker.write_text(datetime.now().isoformat(timespec="seconds"), encoding="utf-8")


def main():
    ap = argparse.ArgumentParser(description="投标看板每日刷新")
    ap.add_argument("--once", action="store_true", default=True, help="单次跑(默认)")
    ap.add_argument("--dry-run", action="store_true", help="只打印,不发请求")
    ap.add_argument("--force", action="store_true", help="忽略今日已跑标记,强制跑")
    ap.add_argument("--no-spawn", action="store_true", help="不触发 p0-watcher")
    args = ap.parse_args()

    cfg = load_config()

    # today 支持测试时强制改
    if cfg.get("today_override"):
        today = datetime.strptime(cfg["today_override"], "%Y-%m-%d").date()
    else:
        today = date.today()

    log(f"========== bid-card-refresh 启动 (today={today}) ==========")

    # 幂等性检查
    if check_idempotency(today, args.force):
        log(f"⏭ 今日已跑过(标记文件存在),跳过(用 --force 强制跑)")
        return

    # 拉 bids
    bids = fetch_bids(cfg)
    if not bids:
        log(f"⚠ 拉取 bids 失败或为空(可能 workbench 未起),退出")
        return
    log(f"  · 拉取 {len(bids)} 条 bids")

    # 步骤 2:重算 priority
    now_str = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
    n_pri_change = n_block = 0
    p0_list = []

    for bid in bids:
        code = bid.get("code")
        if not code:
            continue

        old_pri = bid.get("priority", "P2")
        new_pri = calc_priority(bid, today)

        if new_pri != old_pri:
            n_pri_change += 1
            log(f"  · {code} priority: {old_pri} → {new_pri}")
            if not args.dry_run:
                patch_field(cfg, code, "priority", new_pri)
        if new_pri == "P0":
            p0_list.append(code)

    log(f"  · priority 变化: {n_pri_change} 条")

    # 步骤 3:写 last_sync(给所有 bids,证明看板活着)
    if not args.dry_run:
        for bid in bids:
            code = bid.get("code")
            if code:
                patch_field(cfg, code, "last_sync", now_str)
    log(f"  · last_sync 写入: {len(bids)} 条")

    # 步骤 4:触发 p0-watcher
    if not args.no_spawn:
        spawn_p0_watcher(cfg, p0_list, args.dry_run)

    # 标记今日已跑
    if not args.dry_run:
        mark_done(today)

    log(f"========== bid-card-refresh 完成 ==========\n")


if __name__ == "__main__":
    main()

8风险与回滚
🔴 launchd 不触发

Mac 经常因为系统休眠 / 用户登出导致 launchd 错过定时。
缓解:launchd 配置加 StartCalendarInterval 多日(周一到周五各一行);错过会下次系统唤醒补跑。

🔴 workbench 不起,刷新挂掉

main.py fetch_bids 失败时优雅退出,但 priority 永远不更新。
缓解:日志加 "⚠ 拉取失败" 提示,人工补跑;P1 阶段加"如果连续 N 天失败则告警微信"。

🟡 spawn p0-watcher 失败

workbuddy CLI 未装或路径错,触发不到 P0 告警。
缓解:config.json 的 p0_watcher_cmd 可改;主流程不依赖此步骤,失败仅记日志。

🟡 幂等性误判

同一天多次跑第二次会跳过,如果有 priority 紧急变化就漏了。
缓解:用 --force 强制跑;launchd 默认不传 --force 保持幂等。

🔵 写 last_sync 频繁

每天写 60+ bids 的 last_sync,看板事件流会刷屏。
缓解:事件流按"来源"过滤;last_sync 写入不做 addAiReady(避免噪声)。
