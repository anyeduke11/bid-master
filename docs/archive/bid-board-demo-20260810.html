# 后端 agent / skill 跑完,推看板
import requests

# 看板后端服务(轻量,接收 ingest 事件)
BOARD_URL = "http://127.0.0.1:8765/api/bids/ingest"

def push_bid_update(code: str, **fields):
    requests.post(f"{BOARD_URL}/{code}", json=fields)

# 1. bid-file-review 跑完(任何 skill 都可调用)
push_bid_update("dgyhst2026",
    stage="封标",
    add_ai_ready="100+ 项检查",
    set_block="报价签字"
)

# 2. 订阅 SSE 流(异步接收变化)
import sseclient
events = sseclient.SSEClient("http://127.0.0.1:8765/api/bids/stream")
for event in events:
    # 看板有任何更新都实时收到
    print(event.data)

# 3. 复杂写入(整张卡替换)
requests.put(f"{BOARD_URL}/nfh2026", json={
    "stage": "修订",
    "ai_ready": ["差异点比对", "客户化段落", "BOM v1"],
    "human_todo": ["技术方案重写"],
    "block": ""
})
# 工作台 wb_cli 直接调

# 1. 单字段更新(给东莞加 AI 就绪项)
wb_cli bid update dgyhst2026 \
  --add-ai-ready "100+ 项检查" \
  --set-stage "封标"

# 2. 整张卡覆盖
wb_cli bid write nfh2026 '{
  "stage": "修订",
  "priority": "P1",
  "ai_ready": ["差异点比对", "客户化段落", "BOM v1"],
  "human_todo": ["技术方案重写"],
  "block": ""
}'

# 3. 一次性批量(09:45 看板刷新)
wb_cli bid refresh --all

# 4. 触发 P0 看门狗(任何 stage 变化或 block 出现)
wb_cli bid watch --priority P0 --webhook "lark://im/send?to=duke"

# 5. 任何 skill 都能监听看板事件
wb_cli bid subscribe --on change --do "log_to_cubox"
# 远程 skill / 任何语言都能调(只要能发 HTTP)

# 1. 单条更新
curl -X POST http://127.0.0.1:8765/api/bids/dgyhst2026/ai_ready \
  -H "Content-Type: application/json" \
  -d '{"value": "100+ 项检查"}'

# 2. 整张卡 PATCH
curl -X PATCH http://127.0.0.1:8765/api/bids/nfh2026 \
  -H "Content-Type: application/json" \
  -d '{
    "stage": "修订",
    "ai_ready": ["差异点比对", "BOM v1"],
    "block": ""
  }'

# 3. 订阅(SSE 长连接)
curl -N http://127.0.0.1:8765/api/bids/stream

# 4. 创建新商机
curl -X POST http://127.0.0.1:8765/api/bids \
  -H "Content-Type: application/json" \
  -d '{
    "code": "gxzq260512",
    "client": "国信证券",
    "stage": "跟踪",
    "due_at": "2026-08-25 17:00"
  }'
3同步机制
浏览器内
JS API 同步调用 + 自定义事件 bidboard:change
跨 tab / 跨窗口
BroadcastChannel('bidboard') · 实时推送,无轮询
跨设备 / 跨进程
HTTP API + SSE 流(落到 wb_cli 后端 127.0.0.1:8765)

怎么用这个 demo?① 点右侧按钮看效果;② 点"开第二个 tab"看实时同步;③ 看 API 日志知道每次都调了什么;④ 复制代码贴到你的 skill 里。
